cmake_minimum_required(VERSION 3.24)
project(HerculesDropinRenderer VERSION 0.1.0 LANGUAGES C CXX)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "Use an ignored out-of-source work/build directory")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(DROPIN_BUILD_PROXY "Build Windows x86 DirectDraw forwarding and adapter" ON)
option(DROPIN_BUILD_HOST_SHIM "Compose the MSVC x86 host shim into ddraw.dll" OFF)
option(DROPIN_BUILD_SDL_GPU_DIAGNOSTIC "Build an optional offscreen SDL3 GPU diagnostic" OFF)
include(CTest)

if(MINGW)
    # Generated fixtures must not depend on the caller's compiler DLL PATH.
    add_link_options(-static -static-libgcc -static-libstdc++)
endif()

add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../renderer_contract"
                 "${CMAKE_CURRENT_BINARY_DIR}/renderer_contract")
add_library(dropin_archive_hash STATIC ../src/pc/exe_helpers.c)
# Reuse the recovered hash provider with explicit x86 integer/char semantics in
# the modern build; historical matching retains its independently pinned flags.
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(dropin_archive_hash PRIVATE -fwrapv -fsigned-char)
endif()
add_library(dropin_archive STATIC core/archive_reader.cpp)
target_include_directories(dropin_archive PUBLIC core)
target_link_libraries(dropin_archive PUBLIC dropin_archive_hash)
add_library(dropin_capture STATIC capture/packet_capture.cpp)
target_include_directories(dropin_capture PUBLIC core capture)
target_link_libraries(dropin_capture PUBLIC hercules_renderer_contract)

add_library(dropin_indexed_memory STATIC core/indexed_memory.cpp)
target_include_directories(dropin_indexed_memory PUBLIC core)
target_link_libraries(dropin_indexed_memory PUBLIC hercules_renderer_contract)
add_library(dropin_indexed_reference STATIC backends/indexed_reference.cpp backends/indexed_sprite_reference.cpp)
target_include_directories(dropin_indexed_reference PUBLIC backends)
target_link_libraries(dropin_indexed_reference PUBLIC dropin_indexed_memory dropin_capture)
if(WIN32)
    add_library(dropin_title_resource STATIC core/title_resource.cpp)
    target_include_directories(dropin_title_resource PUBLIC core)
    target_link_libraries(dropin_title_resource PUBLIC dropin_archive dropin_indexed_reference bcrypt)
    add_executable(title_resource_cli tools/title_resource_cli.cpp)
    target_link_libraries(title_resource_cli PRIVATE dropin_title_resource)
endif()
add_executable(indexed_rectangle_cli tools/indexed_rectangle_cli.cpp)
target_link_libraries(indexed_rectangle_cli PRIVATE dropin_indexed_reference)

add_library(dropin_geometry STATIC backends/geometry_diagnostic.cpp)
target_include_directories(dropin_geometry PUBLIC backends)
target_link_libraries(dropin_geometry PUBLIC hercules_renderer_contract)
add_executable(geometry_diagnostic_cli tools/geometry_diagnostic_cli.cpp)
target_link_libraries(geometry_diagnostic_cli PRIVATE dropin_geometry)

if(BUILD_TESTING)
    add_executable(test_archive_reader tests/test_archive_reader.cpp)
    target_link_libraries(test_archive_reader PRIVATE dropin_archive)
    add_test(NAME readonly_archive_reader COMMAND test_archive_reader)
    add_executable(test_indexed_memory tests/test_indexed_memory.cpp)
    target_link_libraries(test_indexed_memory PRIVATE dropin_indexed_memory)
    add_test(NAME owned_indexed_memory COMMAND test_indexed_memory)
    add_executable(test_indexed_reference tests/test_indexed_reference.cpp)
    target_link_libraries(test_indexed_reference PRIVATE dropin_indexed_reference)
    add_test(NAME indexed_reference COMMAND test_indexed_reference)
    add_executable(test_commands tests/test_commands.cpp)
    target_link_libraries(test_commands PRIVATE dropin_capture)
    add_test(NAME command_capture COMMAND test_commands)
    add_executable(test_geometry tests/real_mesh/test_geometry_diagnostic.cpp)
    target_link_libraries(test_geometry PRIVATE dropin_geometry)
    add_test(NAME geometry_diagnostic COMMAND test_geometry)
    add_executable(test_capture_extent ../renderer_contract/tests/test_capture_extent.c)
    target_link_libraries(test_capture_extent PRIVATE hercules_renderer_contract)
    add_test(NAME shared_capture_extent COMMAND test_capture_extent)
    find_package(Python3 3.11 QUIET COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME mesh_runner_validation COMMAND "${Python3_EXECUTABLE}" -B
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/real_mesh/test_runner_validation.py")
    endif()
endif()

if(DROPIN_BUILD_PROXY)
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
        message(FATAL_ERROR "DirectDraw proxy/adapters require Windows x86; configure a Win32/i686 toolchain or disable DROPIN_BUILD_PROXY for offline utilities")
    endif()
    add_library(dropin_host_adapter STATIC proxy/src/engine_table_proxy.cpp)
    target_include_directories(dropin_host_adapter PUBLIC proxy/include)
    target_link_libraries(dropin_host_adapter PUBLIC hercules_renderer_contract)
    add_library(ddraw SHARED proxy/src/ddraw_proxy.cpp proxy/ddraw.def)
    target_include_directories(ddraw PRIVATE proxy/include)
    set_target_properties(ddraw PROPERTIES PREFIX "" OUTPUT_NAME "ddraw")
    if(DROPIN_BUILD_HOST_SHIM)
        if(NOT MSVC)
            message(FATAL_ERROR "The host shim requires MSVC x86 (SEH and native relay)")
        endif()
        target_sources(ddraw PRIVATE proxy/src/host_bootstrap.c
            ../prototypes/host_shim/src/host_shim.c)
        target_include_directories(ddraw PUBLIC ../prototypes/host_shim/include)
        target_compile_definitions(ddraw PRIVATE DROPIN_HOST_SHIM HERCULES_SHIM_BUILD)
        target_compile_options(ddraw PRIVATE /W3 /WX)
        target_link_libraries(ddraw PRIVATE hercules_renderer_contract)
    endif()
    if(MINGW)
        target_link_options(ddraw PRIVATE -Wl,--enable-stdcall-fixup)
    endif()
    if(BUILD_TESTING)
        add_executable(test_proxy tests/proxy_test.cpp tests/mock_ddraw_backend.cpp proxy/src/ddraw_proxy.cpp)
        target_compile_definitions(test_proxy PRIVATE DROPIN_PROXY_TESTING)
        target_include_directories(test_proxy PRIVATE proxy/include)
        target_link_libraries(test_proxy PRIVATE dropin_host_adapter)
        add_test(NAME proxy_forwarding_and_adapter COMMAND test_proxy)
        add_executable(proxy_system_smoke tests/proxy_system_smoke.cpp)
        target_include_directories(proxy_system_smoke PRIVATE proxy/include)
        add_test(NAME proxy_system_forwarding COMMAND proxy_system_smoke "$<TARGET_FILE:ddraw>")
    endif()
endif()

if(DROPIN_BUILD_HOST_SHIM AND NOT DROPIN_BUILD_PROXY)
    message(FATAL_ERROR "DROPIN_BUILD_HOST_SHIM requires DROPIN_BUILD_PROXY")
endif()

if(DROPIN_BUILD_SDL_GPU_DIAGNOSTIC)
    find_package(SDL3 CONFIG REQUIRED)
    add_library(dropin_indexed_gpu STATIC backends/indexed_gpu.cpp)
    target_include_directories(dropin_indexed_gpu PUBLIC backends)
    target_link_libraries(dropin_indexed_gpu PUBLIC SDL3::SDL3 dropin_indexed_reference)
    target_link_libraries(indexed_rectangle_cli PRIVATE dropin_indexed_gpu)
    target_compile_definitions(indexed_rectangle_cli PRIVATE DROPIN_INDEXED_GPU_ENABLED)
    if(WIN32)
        target_link_libraries(title_resource_cli PRIVATE dropin_indexed_gpu)
        target_compile_definitions(title_resource_cli PRIVATE DROPIN_INDEXED_GPU_ENABLED)
        add_executable(hercules_preview tools/hercules_preview.cpp)
        target_link_libraries(hercules_preview PRIVATE dropin_title_resource dropin_indexed_gpu SDL3::SDL3)
        foreach(preview_target title_resource_cli hercules_preview)
            add_custom_command(TARGET ${preview_target} POST_BUILD
                COMMAND "${CMAKE_COMMAND}" -E copy_if_different
                    "$<TARGET_FILE:SDL3::SDL3>" "$<TARGET_FILE_DIR:${preview_target}>")
        endforeach()
    endif()
    add_custom_command(TARGET indexed_rectangle_cli POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "$<TARGET_FILE:SDL3::SDL3>" "$<TARGET_FILE_DIR:indexed_rectangle_cli>")
    add_executable(sdl_gpu_probe backends/sdl_gpu_probe.cpp)
    target_link_libraries(sdl_gpu_probe PRIVATE SDL3::SDL3)
    add_custom_command(TARGET sdl_gpu_probe POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "$<TARGET_FILE:SDL3::SDL3>" "$<TARGET_FILE_DIR:sdl_gpu_probe>")
    if(BUILD_TESTING)
        add_executable(test_indexed_gpu tests/test_indexed_gpu.cpp)
        target_link_libraries(test_indexed_gpu PRIVATE dropin_indexed_gpu)
        add_test(NAME indexed_gpu_comparison COMMAND test_indexed_gpu)
        set_tests_properties(indexed_gpu_comparison PROPERTIES SKIP_RETURN_CODE 77)
        add_test(NAME sdl_gpu_offscreen COMMAND sdl_gpu_probe --diagnostic)
        set_tests_properties(sdl_gpu_offscreen PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
    endif()
endif()

if(BUILD_TESTING)
    get_property(dropin_tests DIRECTORY PROPERTY TESTS)
    set_tests_properties(${dropin_tests} PROPERTIES TIMEOUT 30)
endif()
